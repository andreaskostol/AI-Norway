% Data and methodology

\begin{frame}{Data: A-ordningen}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Norway's mandatory employer-reporting system; universe of formal employment.
  \item Coverage: $\sim 3.1$ million employment records per month, January 2021 through April 2026.
  \item Outcome: \textbf{paid payroll employment}, active relationships with positive cash earnings.
  \item Variables: 4-digit STYRK-08 occupation, age (monthly precision), sector, cash earnings (\emph{kontantlønn}), position percentage, new-job flag.
  \item Decade age groups: 21--30 (early career), 31--40, 41--50, 51--60 (senior).
  \item \textbf{Private sector only}: the same quintile contains different work in the public sector.
  \item Seasonally adjusted (fixed calendar-month factors, 2021--2024); indexed to October 2022 = 1, the month before ChatGPT's public release.
\end{itemize}
\end{frame}

% ---------------------------------------------------------------
\begin{frame}{Exposure measures and quintiles}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Main measure: Eloundou et al.\ (2024) GPT-4 exposure score $\beta$; maps 397 of 407 STYRK-08 codes (97.5\,\% of employment).
  \item Secondary: Handa et al.\ (2025) revealed Claude usage, split into \textbf{automation} and \textbf{augmentation}; maps 352 codes (86.5\,\%).
  \item Rank the mapped codes by exposure; assign each 4-digit code to one of five quintiles on the \emph{occupation} distribution.
  \item Each code counts once, regardless of its employment share. Assignment is fixed throughout the period.
\end{itemize}

\vspace{0.4em}
\footnotesize
Context: Norway ranks 3rd worldwide in population-level AI use (Microsoft 2026); enterprise adoption grew from 11\,\% of firms in 2021 to 30\,\% in 2025 (SSB).
\end{frame}
