% Anthropic (2026): job_exposure measure

\begin{frame}{What the Anthropic 2026 \texttt{job\_exposure} measure tracks}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Time-weighted Claude usage at the occupation level
  \item Penalty applied to directive (automation) conversations
  \item Released by Anthropic in March 2026 as an update to the Anthropic Economic Index
  \item Distinguishes itself from Handa by using time weights rather than conversation shares, and by re-weighting toward augmentation
\end{itemize}
\end{frame}
