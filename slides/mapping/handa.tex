% Handa et al. (2025): Anthropic Economic Index, revealed Claude usage

\begin{frame}{What Handa et al.\ measure}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Sample of Claude.ai conversations classified into O*NET tasks
  \item Occupation exposure $=$ share of conversations matched to that occupation's tasks
  \item Each conversation also classified as:
  \begin{itemize}
    \item \textbf{automation:} directive, user issues a request, accepts the output
    \item \textbf{augmentation:} iterative feedback, refinement
  \end{itemize}
  \item Released as the Anthropic Economic Index (April 2025)
\end{itemize}
\end{frame}
