\begin{frame}{Empirisk spesifikasjon}
\begin{itemize}\setlength{\itemsep}{4pt}
  \item Celler: (yrke $j$ $\times$ aldersgruppe $a$ $\times$ måned $t$), privat sektor, 2021m1--2026m4
  \item Én Poisson-regresjon (PPML) per tiårs aldersgruppe:
\end{itemize}
\[
\log E[\text{antall}_{j,a,t}] = \alpha_{j} + \beta_{t} + \gamma_{q(j),\,k(t)}
\]
\begin{itemize}\setlength{\itemsep}{3pt}
  \item $\alpha_j$: yrkesfaste effekter, fanger opp permanent yrkesstørrelse
  \item $\beta_t$: månedsfaste effekter, fanger opp aldersgruppens samlede tidsbane
  \item $\gamma_{q,k}$: endring i log-punkter for kvintil $q$ ved hendelsestid $k$, relativt til \textbf{Q1} (minst eksponert) og $k = -1$ (oktober 2022)
  \item Standardfeil klynget på yrke
\end{itemize}

\vspace{0.2em}
\footnotesize
\textbf{Hvorfor Poisson?} Antall med en del nuller; $\log(0+x)$ gir skjevhet.\quad
Q3-referanse (median) i backup.\quad\hyperlink{cell-es-q3-detail}{\beamerbutton{Q3-referanse}}
\end{frame}

\begin{frame}{Pretrendene er ikke parallelle. Hvor mye betyr det?}
\small
\begin{itemize}\setlength{\itemsep}{2pt}
  \item Felles tester forkaster parallelle pretrender i alle aldersgrupper; Q5-banen før 2023 svinger 0,11 til 0,18 log-punkter, like mye som bevegelsene etter 2022.
  \item HonestDiD (Rambachan \& Roth 2023): gjennomsnittlig Q5 mot Q1 etter ChatGPT, 21--30 år.
\end{itemize}

\begin{frame}{Hvorfor agentisk KI er et eget vindu}
\textbf{Mekanisme.} Før-agentisk KI: KI hjelper med enkeltsteg i en produksjonskjede, mennesker kontrollerer hvert steg. Agentisk KI: mennesker kontrollerer bare sluttresultatet (Demirer, Horton, Immorlica \& Lucier 2026).

\vspace{0.6em}
\textbf{Når skjedde dette?} Noe vilkårlig, brukskostnaden faller gradvis.
\begin{itemize}\setlength{\itemsep}{2pt}\footnotesize
  \item \textbf{Tidlig 2024.} Devin-demo (mars 2024), Claude 3 Opus (mars 2024), standardiserte API-er for verktøykall. Kjeding teknisk mulig, men på demonstrasjonsnivå.
  \item \textbf{Mai 2025.} OpenAI Codex (\href{https://openai.com/index/introducing-codex/}{16.\ mai 2025}) og Anthropic Claude 4 / Claude Code (\href{https://www.anthropic.com/news/claude-4}{22.\ mai 2025}) gjør flertrinns kodeagenter produksjonsklare.
\end{itemize}
\end{frame}
