\begin{frame}{Hvorfor eksponeringsmålet til Eloundou mfl.}
\hypertarget{after-eloundou}{}
\begin{itemize}\setlength{\itemsep}{10pt}
  \item \textbf{Vanlig brukt standardmål.} Publisert i \textit{Science} (2024), brukt av Brynjolfsson, Chandar \& Chen (2025), senere av Teeselink (2025), Humlum \& Vestergaard (2026), Kauhanen \& Rouvinen (2026) og Lodefalk mfl.\ (2026).
  \item \textbf{Best validering mot observert KI-bruk.} Hos Lindenlaub, Oh, Rodríguez \& Veldkamp (2026, fig.\ 1) har Eloundou mfl.\ $R^2 = 0{,}30$ mot arbeidstakerrapportert bruk, foran Handa (0,26), Felten (0,26) og Webb (0,02).
\end{itemize}
