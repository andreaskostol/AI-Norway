\begin{frame}{Hva Eloundou mfl.\ måler}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Hver O*NET-oppgave er vurdert av mennesker og GPT-4 i tre kategorier:
  \begin{itemize}
    \item[$E_0$] ingen eksponering
    \item[$E_1$] språkmodell alene reduserer tidsbruken med minst 50\,\%
    \item[$E_2$] språkmodell med tilleggsprogramvare reduserer tidsbruken med minst 50\,\%
  \end{itemize}
  \item Oppgaveskår:\quad
        $\beta_{\text{oppgave}} = 0$ ved $E_0$,\;\; $1$ ved $E_1$,\;\; $0{,}5$ ved $E_2$
  \item Yrkesskår: $\beta_{\text{yrke}}$ = snitt av $\beta_{\text{oppgave}}$ der O*NET-oppgaver merket Core teller 2 og Supplemental teller 1
  \item Vi bruker den publiserte yrkesfilen (\texttt{occ\_level.csv}); den avviker litt fra uvektede oppgavesnitt der yrket har Supplemental-oppgaver
\end{itemize}
\end{frame}
