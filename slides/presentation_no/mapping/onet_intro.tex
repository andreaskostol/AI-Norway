% O*NET-datastrukturen: bakgrunn for eksponeringsmålene

\begin{frame}{O*NET: den amerikanske yrkesdatabasen under tre av de fire målene}
\begin{itemize}\setlength{\itemsep}{6pt}
  \item Vedlikeholdes av US Department of Labor; dekker rundt 1\,000 yrker på SOC 2018-koder (O*NET-SOC 2019-taksonomien)
  \item Hvert yrke har strukturert innhold:
  \begin{itemize}
    \item \textbf{Oppgaver:} avgrensede aktiviteter (rundt 19\,000 totalt; farmasøyter: 21 oppgaver, f.eks.\ «review prescriptions for accuracy»). Merket \emph{Core} eller \emph{Supplemental}; ingen numerisk vekt.
    \item \textbf{Evner:} 52 underliggende egenskaper (f.eks.\ deduktiv resonnering, fingerferdighet). Skåret på utbredelse og viktighet per yrke.
    \item I tillegg: ferdigheter, kunnskap, arbeidsaktiviteter og arbeidskontekst (brukes ikke av målene her).
  \end{itemize}
  \item Eloundou mfl.\ og Handa bygger på \textbf{oppgaver}; Felten bygger på \textbf{evner}; Anthropic 2026 er på yrkesnivå.
\end{itemize}
\end{frame}
